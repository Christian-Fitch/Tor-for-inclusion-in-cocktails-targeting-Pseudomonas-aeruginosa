annotation pipeline for the two phages - CPL00272 and CPL00272 variant isolated on DFI10.1

source ~/.bashrc

#for the first one CPL00272, using the PAO1 genome as a reference 
conda activate bbmap

bbmap.sh \
ref=PAO1.fasta \
nodisk=t \
minid=0.95 maxindel=3 bwr=0.16 bw=12 quickmatch fast minhits=2 \
qtrim=rl trimq=10 untrim \
in1=11566_CPL00272_CPL_S16_R1_001_fastp.fastq.gz \
in2=11566_CPL00272_CPL_S16_R2_001_fastp.fastq.gz \
ordered=t \
outu=unmapped.fq.gz 2>&1 | tee -a 272_CPL.log

#6% mapped

clumpify.sh -da in=unmapped.fq.gz out=unmapped.clumped.fq.gz dedupe optical 2>&1 | tee -a 272_CPL.log
filterbytile.sh -da in=unmapped.clumped.fq.gz out=unmapped.filtered_by_tile.fq.gz 2>&1 | tee -a 272_CPL.log
bbduk.sh in=unmapped.filtered_by_tile.fq.gz out=unmapped.trimmed.fq.gz ktrim=r k=23 mink=11 hdist=1 tbo tpe minlen=50 ref=adapters ftm=5 ordered 2>&1 | tee -a 272_CPL.log
bbduk.sh in=unmapped.trimmed.fq.gz out=unmapped.filtered.fq.gz k=31 ref=artifacts,phix ordered cardinality 2>&1 | tee -a 272_CPL.log
tadpole.sh in=unmapped.filtered.fq.gz out=unmapped.ec.fq.gz mode=correct k=62 ecc ecco merge=t 2>&1 | tee -a 272_CPL.log
reformat.sh in=unmapped.ec.fq.gz out1=272_CPL.fwd.ec.fq.gz out2=272_CPL.rev.ec.fq.gz

rm unmapped.fq.gz
rm unmapped.clumped.fq.gz
rm unmapped.filtered_by_tile.fq.gz
rm unmapped.trimmed.fq.gz
rm unmapped.filtered.fq.gz
rm unmapped.ec.fq.gz

#assmebly de novo using shovill

conda activate shovill

shovill --outdir shovill \
--R1 272_CPL.fwd.ec.fq.gz \
--R2 272_CPL.rev.ec.fq.gz \
--mincov 20 \
--cpus 64 \
--ram 120 \
--minlen 10000 \
--force  2>&1 | tee -a 272_CPL.log

# a beautiful single contig ;)

conda activate checkv
checkv end_to_end shovill/contigs.fa checkv -t 64 -d /home/artic/bens_toys/dbs/checkv-db-v1.1/ 2>&1 | tee -a 272_CPL.log
cat checkv/quality_summary.tsv

conda activate pharokka
pharokka.py -i checkv/viruses.fna -o 272_CPL-pharokka -d /mnt/773be256-7089-43eb-a736-be2e5373be02/CPL/dbs/pharokka-1.2.0 -t 64 -p 272_CPL -l 272_CPL
#terminase large subunit found at 38056 on the positive strand 

pharokka.py -i checkv/viruses.fna -o 272_CPLreordered-pharokka -d /mnt/773be256-7089-43eb-a736-be2e5373be02/CPL/dbs/pharokka-1.2.0 -t 64 -p 272_CPL -l 272_CPL --terminase --terminase_strand pos --terminase_start 38056 -f

pharokka_plotter.py -p 272_CPL -i checkv/viruses.fna -n 272_CPL_plot -o 272_CPL-pharokka --truncate 20 --interval 10000 --plot_title 'Pseudomonas phage 272_CPL' --dpi 300
